\heading{Two-Spin States and Coefficient Matrices}
\textit{Examples 01 and 02}

\subsection*{States and normalization}
Begin with
\begin{equation}
\ket{\uparrow}=\begin{pmatrix}1\\0\end{pmatrix},\qquad
\ket{\downarrow}=\begin{pmatrix}0\\1\end{pmatrix},\qquad
\ket{+}=\frac{\ket{\uparrow}+\ket{\downarrow}}{\sqrt2}.
\end{equation}
In the order $\ket{\uparrow\uparrow},\ket{\uparrow\downarrow},
\ket{\downarrow\uparrow},\ket{\downarrow\downarrow}$, the two states are
\begin{align}
\ket{\psi_{\rm prod}}
&=\ket{+}\otimes\ket{\uparrow}
=\frac{\ket{\uparrow\uparrow}+\ket{\downarrow\uparrow}}{\sqrt2}
=\frac1{\sqrt2}\vfour{1}{0}{1}{0},\\
\ket{\psi_{\rm Bell}}
&=\frac{\ket{\uparrow\uparrow}+\ket{\downarrow\downarrow}}{\sqrt2}
=\frac1{\sqrt2}\vfour{1}{0}{0}{1}.
\end{align}
Both are normalized:
\begin{equation}
\langle\psi|\psi\rangle=\sum_{ij}|C_{ij}|^2
=\frac12+\frac12=1.
\end{equation}
In code, this is \texttt{np.vdot(state, state)}.

\subsection*{Reshaping and separability}
Write
\begin{equation}
\ket{\psi}=\sum_{ij}C_{ij}\ket{i}\otimes\ket{j},\qquad
\vfour{C_{\uparrow\uparrow}}{C_{\uparrow\downarrow}}
{C_{\downarrow\uparrow}}{C_{\downarrow\downarrow}}
\xrightarrow{\text{reshape}}
C=\mtwo{C_{\uparrow\uparrow}}{C_{\uparrow\downarrow}}
{C_{\downarrow\uparrow}}{C_{\downarrow\downarrow}}.
\end{equation}
This is \texttt{state.reshape(2, 2)}. For a product state,
\begin{align}
\ket{a}\otimes\ket{b}
&=\left(\sum_i a_i\ket{i}\right)\otimes
\left(\sum_j b_j\ket{j}\right)=\sum_{ij}a_i b_j\ket{ij},\\
C_{ij}&=a_i b_j,\qquad C=ab^{\mathsf T},\qquad\operatorname{rank}C=1.
\end{align}
Conversely, a nonzero rank-one $C$ factors as $ab^{\mathsf T}$, giving a product state.
The outer product is \texttt{np.outer(a, b)}, without conjugation.
\begin{equation}
C_{\rm prod}=\frac1{\sqrt2}\mtwo{1}{0}{1}{0}
=\frac1{\sqrt2}\begin{pmatrix}1\\1\end{pmatrix}\begin{pmatrix}1&0\end{pmatrix},
\qquad C_{\rm Bell}=\frac1{\sqrt2}\mtwo{1}{0}{0}{1}.
\end{equation}
Their ranks are one and two: the product state is separable, and the Bell state is entangled. For a pure bipartite state, this is precisely what the rank of $C$ distinguishes.
